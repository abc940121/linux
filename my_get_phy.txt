#include <unistd.h>

#define SYS_MY_GET_PHYSICAL_ADDRESSES 548

struct va_pa_struct{
	unsigned long va;
	unsigned long pa;
};

unsigned long my_get_physical_addresses(void *virtual_addr){
	struct va_pa_struct data;
	data.va = (unsigned long)virtual_addr;
	data.pa = 0;
	long status = syscall(SYS_MY_GET_PHYSICAL_ADDRESSES, &data);
	if (status != 0 )return 0;
	return data.pa;
}

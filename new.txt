struct va_pa_struct{
    unsigned long va;
    unsigned long pa;
};

static unsigned long do_one_translation(unsigned long vaddr){
    pgd_t *pgd;
    p4d_t *p4d;
    pud_t *pud;
    pmd_t *pmd;
    pte_t *ptep, pte;
    unsigned long paddr = 0;
    struct mm_struct *mm = current->mm;

    if (!mm)
        return 0;

    pgd = pgd_offset(mm, vaddr);
    if (pgd_none(*pgd) || pgd_bad(*pgd))
        return 0;

    p4d = p4d_offset(pgd, vaddr);
    if (p4d_none(*p4d) || p4d_bad(*p4d))
        return 0;

    pud = pud_offset(p4d, vaddr);
    if (pud_none(*pud) || pud_bad(*pud))
        return 0;

    pmd = pmd_offset(pud, vaddr);
    if (pmd_none(*pmd) || pmd_bad(*pmd))
        return 0;

    ptep = pte_offset_map(pmd, vaddr);
    if (!ptep)
        return 0;

    pte = *ptep;

    if (pte_present(pte)) {
        paddr = (pte_pfn(pte) << PAGE_SHIFT) | (vaddr & ~PAGE_MASK);
    }

    pte_unmap(ptep);
    return paddr;
}

SYSCALL_DEFINE1(my_get_physical_addresses,
                struct va_pa_struct __user *, user_data)
{
    struct va_pa_struct k_data;

    if (copy_from_user(&k_data, user_data,
                       sizeof(struct va_pa_struct))) {
        return -EFAULT;
    }

    k_data.pa = do_one_translation(k_data.va);

    if (copy_to_user(user_data, &k_data,
                     sizeof(struct va_pa_struct))) {
        return -EFAULT;
    }
    return 0;
}
